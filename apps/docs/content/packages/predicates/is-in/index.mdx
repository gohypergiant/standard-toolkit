# Array Membership Predicates

Curried predicate functions for testing whether values exist in arrays using strict equality comparison.

## Usage

```typescript
import { isIn, isNotIn } from '@accelint/predicates';

const isValidStatus = isIn(['pending', 'approved', 'rejected']);
isValidStatus('pending');  // true
isValidStatus('unknown');  // false

const isInvalidStatus = isNotIn(['pending', 'approved', 'rejected']);
isInvalidStatus('unknown');  // true
isInvalidStatus('pending');  // false
```

## isIn

Creates a predicate that determines if a value exists in a given array.

### Reference

```typescript
function isIn<T>(array: T[]): (value: T) => boolean
```

#### Parameters

| Parameter | Type | Description |
|-----------|------|-------------|
| `array` | `T[]` | Array to search within |

#### Returns

Curried function that accepts a value and returns `true` if the value exists in the array (using strict equality).

### Examples

#### Example: Status validation

```typescript
import { isIn } from '@accelint/predicates';

const VALID_STATUSES = ['draft', 'published', 'archived'];
const isValidStatus = isIn(VALID_STATUSES);

function validatePost(status: string): boolean {
  return isValidStatus(status);
}

validatePost('published'); // true
validatePost('deleted');   // false
```

#### Example: Permission checking

```typescript
import { isIn } from '@accelint/predicates';

const userPermissions = ['read', 'write', 'delete'];
const hasPermission = isIn(userPermissions);

hasPermission('read');   // true
hasPermission('admin');  // false
```

#### Example: Filtering valid IDs

```typescript
import { isIn } from '@accelint/predicates';

const allowedIds = [101, 102, 103, 104];
const isAllowedId = isIn(allowedIds);

const requestIds = [101, 105, 102, 106];
const validRequests = requestIds.filter(isAllowedId);
// [101, 102]
```

## isNotIn

Creates a predicate that determines if a value does **not** exist in a given array.

### Reference

```typescript
function isNotIn<T>(array: T[]): (value: T) => boolean
```

#### Parameters

| Parameter | Type | Description |
|-----------|------|-------------|
| `array` | `T[]` | Array to search within |

#### Returns

Curried function that accepts a value and returns `true` if the value does **not** exist in the array.

### Examples

#### Example: Filtering unauthorized requests

```typescript
import { isNotIn } from '@accelint/predicates';

const blockedUsers = [101, 205, 399];
const isNotBlocked = isNotIn(blockedUsers);

const incomingRequests = [101, 102, 205, 300];
const allowed = incomingRequests.filter(isNotBlocked);
// [102, 300]
```

#### Example: Excluding reserved keywords

```typescript
import { isNotIn } from '@accelint/predicates';

const RESERVED_WORDS = ['class', 'function', 'return', 'if', 'else'];
const isNotReserved = isNotIn(RESERVED_WORDS);

function isValidIdentifier(name: string): boolean {
  return isNotReserved(name) && /^[a-zA-Z_]\w*$/.test(name);
}

isValidIdentifier('myVariable'); // true
isValidIdentifier('class');      // false
```

#### Example: Finding new items

```typescript
import { isNotIn } from '@accelint/predicates';

const existingIds = [1, 2, 3, 4, 5];
const isNewId = isNotIn(existingIds);

const incomingIds = [4, 5, 6, 7];
const newItems = incomingIds.filter(isNewId);
// [6, 7]
```

## Real-World Example: Role-based access control

```typescript
import { isIn, isNotIn } from '@accelint/predicates';

const ADMIN_ROLES = ['admin', 'super-admin'];
const RESTRICTED_ACTIONS = ['delete-user', 'modify-permissions'];

const isAdmin = isIn(ADMIN_ROLES);
const isNotRestricted = isNotIn(RESTRICTED_ACTIONS);

function canPerformAction(userRole: string, action: string): boolean {
  if (isAdmin(userRole)) {
    return true; // Admins can do anything
  }
  
  return isNotRestricted(action);
}

canPerformAction('admin', 'delete-user');     // true
canPerformAction('user', 'view-dashboard');   // true
canPerformAction('user', 'delete-user');      // false
```

> **Good to know:** These predicates use `Array.prototype.includes()` internally, which uses strict equality (`===`). For deep equality or custom comparison logic, use a different approach or preprocess your data.
