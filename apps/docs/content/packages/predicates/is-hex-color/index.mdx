# Color Format Predicates

Type guards for validating various color format representations used in web and graphics applications.

## Usage

```typescript
import {
  isHexColor,
  isCssRgbaString,
  isCssRgbaObject,
  isRgba255Tuple,
  isValid255Channel
} from '@accelint/predicates';

isHexColor('#FF8040');          // true
isCssRgbaString('rgba(255, 128, 64, 0.5)'); // true
isCssRgbaObject({ r: 255, g: 128, b: 64, a: 1 }); // true
isRgba255Tuple([255, 128, 64, 255]); // true
isValid255Channel(128);         // true
```

## isHexColor

Validates hex color strings with optional alpha channel.

### Reference

```typescript
function isHexColor(value: unknown): value is string
```

Supports formats: `#RGB`, `#RGBA`, `#RRGGBB`, `#RRGGBBAA` (hash optional)

### Examples

```typescript
import { isHexColor } from '@accelint/predicates/is-hex-color';

isHexColor('#FF8040');    // true
isHexColor('#F84');       // true
isHexColor('FF8040');     // true (hash optional)
isHexColor('#FF8040FF');  // true (with alpha)
isHexColor('rgb(255, 128, 64)'); // false
```

## isCssRgbaString

Validates CSS rgb/rgba string format (legacy and modern syntax).

### Reference

```typescript
function isCssRgbaString(value: unknown): value is string
```

Supports:
- Legacy: `rgb(255, 128, 64)`, `rgba(255, 128, 64, 0.5)`
- Modern: `rgb(255 128 64)`, `rgb(255 128 64 / 0.5)`

### Examples

```typescript
import { isCssRgbaString } from '@accelint/predicates/is-css-rgba-string';

// Legacy syntax
isCssRgbaString('rgba(255, 128, 64, 0.5)'); // true
isCssRgbaString('rgb(255, 128, 64)');       // true

// Modern syntax (CSS Color Module Level 4)
isCssRgbaString('rgb(255 128 64 / 50%)');   // true
isCssRgbaString('rgb(255 128 64)');         // true

isCssRgbaString('#FF8040');                 // false
```

## isCssRgbaObject

Validates CSS-style RGBA color object (React Aria Components format).

### Reference

```typescript
type CssRgbaObject = {
  readonly r: number; // 0-255
  readonly g: number; // 0-255
  readonly b: number; // 0-255
  readonly a: number; // 0-1
};

function isCssRgbaObject(value: unknown): value is CssRgbaObject
```

RGB channels: 0-255, Alpha channel: 0-1

### Examples

```typescript
import { isCssRgbaObject } from '@accelint/predicates/is-css-rgba-object';

isCssRgbaObject({ r: 255, g: 128, b: 64, a: 1 });    // true
isCssRgbaObject({ r: 255, g: 128, b: 64, a: 0.5 });  // true
isCssRgbaObject({ r: 255, g: 128, b: 64 });          // false (missing alpha)
isCssRgbaObject([255, 128, 64, 255]);                // false (array, not object)
isCssRgbaObject({ r: 256, g: 128, b: 64, a: 1 });    // false (r out of range)
```

## isRgba255Tuple

Validates RGBA tuple where all channels are 0-255 (deck.gl format).

### Reference

```typescript
type Rgba255Tuple = readonly [
  r: number, // 0-255
  g: number, // 0-255
  b: number, // 0-255
  a: number, // 0-255
];

function isRgba255Tuple(value: unknown): value is Rgba255Tuple
```

### Examples

```typescript
import { isRgba255Tuple } from '@accelint/predicates/is-rgba-255-tuple';

isRgba255Tuple([255, 128, 64, 255]);  // true
isRgba255Tuple([255, 128, 64, 128]);  // true (alpha as 0-255)
isRgba255Tuple([255, 128, 64]);       // false (missing alpha)
isRgba255Tuple('rgba(255, 128, 64, 1)'); // false (string, not tuple)
isRgba255Tuple([256, 128, 64, 255]);  // false (r out of range)
```

## isValid255Channel

Validates a single color channel value (0-255).

### Reference

```typescript
function isValid255Channel(n: number): boolean
```

### Examples

```typescript
import { isValid255Channel } from '@accelint/predicates/is-valid-255-channel';

isValid255Channel(0);    // true
isValid255Channel(128);  // true
isValid255Channel(255);  // true
isValid255Channel(256);  // false
isValid255Channel(-1);   // false
isValid255Channel(NaN);  // false
```
