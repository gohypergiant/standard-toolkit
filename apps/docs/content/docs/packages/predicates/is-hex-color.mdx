Validates hex color strings with optional alpha channel.

## Usage

```typescript
import { isHexColor } from '@accelint/predicates/is-hex-color';

isHexColor('#FF8040');    // true
isHexColor('#F84');       // true
isHexColor('FF8040');     // true (hash optional)
isHexColor('#FF8040FF');  // true (with alpha)
isHexColor('rgb(255, 128, 64)'); // false
```

## Reference

```typescript
function isHexColor(value: unknown): value is string
```

Supports formats: `#RGB`, `#RGBA`, `#RRGGBB`, `#RRGGBBAA` (hash optional)

## Examples

```typescript
import { isHexColor } from '@accelint/predicates/is-hex-color';

// Standard 6-digit hex
isHexColor('#FF8040');    // true
isHexColor('FF8040');     // true (hash optional)

// Short 3-digit hex
isHexColor('#F84');       // true
isHexColor('F84');        // true

// With alpha channel
isHexColor('#FF8040FF');  // true (8-digit)
isHexColor('#F84F');      // true (4-digit)

// Invalid formats
isHexColor('rgb(255, 128, 64)'); // false
isHexColor('#GG8040');    // false (invalid hex chars)
isHexColor('#FF804');     // false (wrong length)
```
