## Usage

```typescript
import {
  datasetId,
  datasetName,
  datasetTable,
  datasetServiceUrls,
  datasetGeometryProperty,
} from '@accelint/dataset';

// Extract root properties
const id = datasetId(dataset);
const name = datasetName(dataset);

// Extract metadata properties (composed lenses)
const tableName = datasetTable(dataset);
const urls = datasetServiceUrls(dataset);

// Type-safe property extraction with generics
type Feature = { id: number; location: Point };
const geomKey = datasetGeometryProperty<Feature>(dataset); // 'location'
const coords = feature[geomKey];
```

## Reference

The package exports two categories of lens functions: **Root Property Lenses** for direct dataset properties and **Composed Lenses** for nested metadata properties.

## Examples

### Example: Building a dataset display component

```typescript
import {
  datasetId,
  datasetName,
  datasetDescription,
  datasetServiceType,
  datasetVisible,
} from '@accelint/dataset';

function DatasetCard({ dataset }: { dataset: AnyDataset }) {
  const id = datasetId(dataset);
  const name = datasetName(dataset);
  const description = datasetDescription(dataset);
  const serviceType = datasetServiceType(dataset);
  const isVisible = datasetVisible(dataset);

  return (
    <div data-dataset-id={id}>
      <h2>{name}</h2>
      <p>{description}</p>
      <span>Service: {serviceType}</span>
      {isVisible && <span>Currently visible</span>}
    </div>
  );
}
```

### Example: Building service endpoint URLs

```typescript
import {
  datasetServiceUrls,
  datasetServiceVersion,
  datasetServiceLayer,
  datasetTable,
} from '@accelint/dataset';

function buildWFSUrl(dataset: AnyDataset): string {
  const urls = datasetServiceUrls(dataset);
  const version = datasetServiceVersion(dataset);
  const layer = datasetServiceLayer(dataset);
  const table = datasetTable(dataset);

  const baseUrl = urls?.[0] || '';
  const queryParams = new URLSearchParams({
    service: 'WFS',
    version: version || '2.0.0',
    request: 'GetFeature',
    typeName: layer || table,
    outputFormat: 'application/json',
  });

  return `${baseUrl}?${queryParams}`;
}
```

### Example: Type-safe feature property access

```typescript
import { datasetIdProperty, datasetGeometryProperty } from '@accelint/dataset';

type WeatherFeature = {
  // biome-ignore lint/style/useNamingConvention: API response format
  station_id: string;
  temperature: number;
  humidity: number;
  location: [number, number];
};

function extractFeatureData(
  dataset: AnyDataset,
  feature: WeatherFeature
) {
  // Type-safe property extraction
  const idKey = datasetIdProperty<WeatherFeature>(dataset); // 'station_id'
  const geomKey = datasetGeometryProperty<WeatherFeature>(dataset); // 'location'

  return {
    id: feature[idKey],
    coordinates: feature[geomKey],
    temperature: feature.temperature,
  };
}
```

### Example: Configuring data fetching

```typescript
import {
  datasetMaxRequests,
  datasetRefetchInterval,
  datasetBatchSize,
  datasetDefaultFields,
} from '@accelint/dataset';

function createFetchConfig(dataset: AnyDataset) {
  const maxRequests = datasetMaxRequests(dataset) || 6;
  const refetchInterval = datasetRefetchInterval(dataset);
  const batchSize = datasetBatchSize(dataset) || 10000;
  const fields = datasetDefaultFields(dataset);

  return {
    concurrency: maxRequests,
    polling: refetchInterval ? { interval: refetchInterval } : null,
    pagination: { pageSize: batchSize },
    fields: fields.join(','),
  };
}
```

### Example: Handling backend-specific configuration

```typescript
import {
  datasetBackend,
  datasetVendorParams,
  datasetFilterDialect,
} from '@accelint/dataset';

function buildBackendConfig(dataset: AnyDataset) {
  const backend = datasetBackend(dataset);
  const vendorParams = datasetVendorParams(dataset);
  const filterDialect = datasetFilterDialect(dataset);

  if (backend === 'geoserver' && vendorParams) {
    return {
      backend,
      formatOptions: vendorParams.formatOptions,
      viewparams: vendorParams.viewparams,
      filterFormat: filterDialect || 'cql',
    };
  }

  return { backend, filterFormat: filterDialect || 'cql' };
}
```

### Example: Rendering field-based visualizations

```typescript
import {
  datasetFields,
  datasetPresentationTypes,
  datasetDataType,
} from '@accelint/dataset';

function selectVisualization(dataset: AnyDataset, fieldName: string) {
  const fields = datasetFields(dataset);
  const presentationTypes = datasetPresentationTypes(dataset);
  const dataType = datasetDataType(dataset);

  const field = fields.find(f => f.id === fieldName);
  const availableTypes = presentationTypes[fieldName];

  if (!field || !availableTypes) {
    return null;
  }

  // Choose appropriate visualization based on field type and data format
  if (dataType === 'ARROW' && availableTypes.includes('point')) {
    return { type: 'point', field: field.id, label: field.label };
  }

  return { type: availableTypes[0], field: field.id, label: field.label };
}
```

> **Good to know:** All lens functions are pure and do not modify the input dataset. Optional properties return `undefined` when not present. The generic type parameters on `datasetIdProperty` and `datasetGeometryProperty` provide compile-time type safety for feature property access.
