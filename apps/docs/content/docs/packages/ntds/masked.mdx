Numeric string identifiers for NTDS tactical symbols in spritesheet format. These constants are used for efficient rendering in Common Operating Picture (COP) applications where sprite-based rendering is preferred over individual SVG components.

## Usage

```typescript
import { AIR_FRIEND, SURFACE_HOSTILE } from '@accelint/ntds/spritesheets/masked';

// Reference sprites by ID in COP rendering engine
const friendlyAircraftSprite = spritesheet.getSprite(AIR_FRIEND);
const hostileShipSprite = spritesheet.getSprite(SURFACE_HOSTILE);
```

## Reference

All constants are numeric string identifiers generated using CRC-32 hashing:

```typescript
const CONSTANT_NAME: string; // e.g., '978699787'
```

Each constant represents a unique sprite frame in the NTDS spritesheet. The numeric IDs are deterministic and stable across builds.

> **Good to know:** These constants are generated by the `smeegl` tool with CRC-32 decimal encoding. The IDs remain consistent as long as the source SVG files don't change.

## Examples

### Example: Sprite lookup in rendering engine

```typescript
import { AIR_FRIEND, SURFACE_HOSTILE } from '@accelint/ntds/spritesheets/masked';

class CopRenderer {
  private spritesheet: Spritesheet;

  renderEntity(entity: Entity): void {
    const spriteId = this.getSpriteId(entity);
    const sprite = this.spritesheet.getSprite(spriteId);
    
    this.drawSprite(sprite, entity.x, entity.y);
  }

  private getSpriteId(entity: Entity): string {
    if (entity.domain === 'air' && entity.affiliation === 'friend') {
      return AIR_FRIEND;
    }
    if (entity.domain === 'surface' && entity.affiliation === 'hostile') {
      return SURFACE_HOSTILE;
    }
    // ... more mappings
  }
}
```

### Example: Sprite ID mapping table

```typescript
import {
  AIR_FRIEND,
  AIR_HOSTILE,
  AIR_UNKNOWN,
  SURFACE_FRIEND,
  SURFACE_HOSTILE,
  SURFACE_UNKNOWN,
} from '@accelint/ntds/spritesheets/masked';

type Domain = 'air' | 'surface';
type Affiliation = 'friend' | 'hostile' | 'unknown';

const SPRITE_MAP: Record<Domain, Record<Affiliation, string>> = {
  air: {
    friend: AIR_FRIEND,
    hostile: AIR_HOSTILE,
    unknown: AIR_UNKNOWN,
  },
  surface: {
    friend: SURFACE_FRIEND,
    hostile: SURFACE_HOSTILE,
    unknown: SURFACE_UNKNOWN,
  },
};

function getSpriteId(domain: Domain, affiliation: Affiliation): string {
  return SPRITE_MAP[domain][affiliation];
}
```

### Example: Canvas rendering with sprites

```typescript
import { AIR_FRIEND } from '@accelint/ntds/spritesheets/masked';

function renderTacticalIcon(
  ctx: CanvasRenderingContext2D,
  spritesheet: HTMLImageElement,
  spriteData: SpritesheetData,
  x: number,
  y: number
): void {
  const sprite = spriteData.frames[AIR_FRIEND];
  
  ctx.drawImage(
    spritesheet,
    sprite.frame.x,
    sprite.frame.y,
    sprite.frame.w,
    sprite.frame.h,
    x,
    y,
    sprite.frame.w,
    sprite.frame.h
  );
}
```

### Example: WebGL sprite rendering

```typescript
import { SURFACE_GROUP_HOSTILE } from '@accelint/ntds/spritesheets/masked';

interface SpriteInstance {
  spriteId: string;
  x: number;
  y: number;
  scale: number;
}

class WebGLCopRenderer {
  private spritesheet: WebGLTexture;
  private spriteData: SpritesheetData;

  renderBatch(instances: SpriteInstance[]): void {
    const vertices = instances.flatMap((instance) => {
      const sprite = this.spriteData.frames[instance.spriteId];
      return this.createQuad(sprite, instance.x, instance.y, instance.scale);
    });
    
    this.drawVertices(vertices);
  }
}

// Usage
const hostileFleet: SpriteInstance[] = [
  { spriteId: SURFACE_GROUP_HOSTILE, x: 100, y: 200, scale: 1.0 },
  { spriteId: SURFACE_GROUP_HOSTILE, x: 150, y: 220, scale: 1.0 },
];
```

### Example: Dynamic affiliation state changes

```typescript
import {
  SURFACE_UNKNOWN,
  SURFACE_SUSPECT,
  SURFACE_HOSTILE,
} from '@accelint/ntds/spritesheets/masked';

class TrackedEntity {
  private currentSpriteId: string;

  constructor() {
    this.currentSpriteId = SURFACE_UNKNOWN;
  }

  updateAffiliation(confidence: number): void {
    if (confidence > 0.8) {
      this.currentSpriteId = SURFACE_HOSTILE;
    } else if (confidence > 0.5) {
      this.currentSpriteId = SURFACE_SUSPECT;
    } else {
      this.currentSpriteId = SURFACE_UNKNOWN;
    }
  }

  getSpriteId(): string {
    return this.currentSpriteId;
  }
}
```

### Example: Sprite preloading

```typescript
import * as SPRITES from '@accelint/ntds/spritesheets/masked';

class SpritesheetManager {
  private loadedSprites = new Set<string>();

  async preloadCommonSprites(): Promise<void> {
    const commonSprites = [
      SPRITES.AIR_FRIEND,
      SPRITES.AIR_HOSTILE,
      SPRITES.SURFACE_FRIEND,
      SPRITES.SURFACE_HOSTILE,
      SPRITES.SUB_SURFACE_UNKNOWN,
    ];

    await Promise.all(
      commonSprites.map((spriteId) => this.loadSprite(spriteId))
    );
  }

  private async loadSprite(spriteId: string): Promise<void> {
    // Load sprite data
    this.loadedSprites.add(spriteId);
  }
}
```

### Example: Type-safe sprite selection

```typescript
import type * as SPRITES from '@accelint/ntds/spritesheets/masked';

type SpriteId = typeof SPRITES[keyof typeof SPRITES];

function renderSprite(spriteId: SpriteId, x: number, y: number): void {
  // Type-safe sprite ID
  console.log(`Rendering sprite ${spriteId} at (${x}, ${y})`);
}

// Valid
renderSprite(SPRITES.AIR_FRIEND, 10, 20);

// Type error - invalid sprite ID
// renderSprite('invalid-id', 10, 20);
```
