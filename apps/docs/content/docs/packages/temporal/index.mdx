## Installation

```bash
pnpm add @accelint/temporal
```

## Usage

```typescript
// Import from root (all exports)
import { setClockInterval, setClockTimeout } from '@accelint/temporal';

// Import from sub-modules
import { setClockInterval, setClockTimeout } from '@accelint/temporal/timers';
```

## Available Exports

### Timers

| Function | Description |
|----------|-------------|
| [setClockInterval](./set-clock-interval.mdx) | Execute a callback repeatedly at specified intervals, starting at the next clock second boundary |
| [setClockTimeout](./set-clock-timeout.mdx) | Execute a callback once after a specified delay, starting from the next clock second boundary |

## Examples

### Example: Clock display

```typescript
import { setClockInterval } from '@accelint/temporal';

function startClock(element: HTMLElement) {
  const cleanup = setClockInterval(() => {
    element.textContent = new Date().toLocaleTimeString();
  }, 1000);
  
  return cleanup;
}

// Start the clock
const cleanup = startClock(document.getElementById('clock'));

// Stop when done
cleanup();
```

### Example: Scheduled task

```typescript
import { setClockTimeout } from '@accelint/temporal';

function scheduleNotification(message: string, delayMs: number) {
  const cleanup = setClockTimeout(() => {
    console.log(message);
  }, delayMs);
  
  return cleanup;
}

// Schedule for 5 seconds from next clock second
const cancel = scheduleNotification('Task complete!', 5000);

// Cancel if needed
cancel();
```
